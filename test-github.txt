/* Service Worker — اجرای آفلاین اپ کلینیک (نسخه ۱) */
const CACHE = 'clinic-pwa-v1';
const FILES = [
  './',
  './index.html',
  './manifest.webmanifest',
  './icons/icon-192.png',
  './icons/icon-512.png',
  './icons/apple-touch-icon.png'
];

self.addEventListener('install', (e) => {
  e.waitUntil(
    caches.open(CACHE).then((c) => c.addAll(FILES)).then(() => self.skipWaiting())
  );
});

self.addEventListener('activate', (e) => {
  e.waitUntil(
    caches.keys().then((keys) =>
      Promise.all(keys.filter((k) => k !== CACHE).map((k) => caches.delete(k)))
    ).then(() => self.clients.claim())
  );
});

self.addEventListener('fetch', (e) => {
  if (e.request.method !== 'GET') return;
  e.respondWith(
    caches.match(e.request, { ignoreSearch: true }).then((hit) => {
      if (hit) return hit;
      return fetch(e.request).then((res) => {
        const copy = res.clone();
        caches.open(CACHE).then((c) => c.put(e.request, copy)).catch(() => {});
        return res;
      }).catch(() => caches.match('./index.html'));
    })
  );
});

راهنمای نصب اپ کلینیک دندانپزشکی (دسکتاپ و اندروید)
=====================================================

این پوشه یک «وب‌اپ نصبی» (PWA) است: بدون اینترنت کار می‌کند،
آیکون اختصاصی دارد و تمام‌صفحه مثل اپ واقعی اجرا می‌شود.

فایل‌ها:
- index.html : خود برنامه (شیفت + پذیرش + تاریخ شمسی)
- manifest.webmanifest : مشخصات اپ و آیکون
- sw.js : حالت آفلاین
- icons/ : آیکون‌ها
- start-clinic.bat : اجرای سریع روی ویندوز

-----------------------------------------------------
۱) دسکتاپ ویندوز (Chrome یا Edge)
-----------------------------------------------------
روش سریع (بدون اینترنت):
۱. پوشه را در کامپیوتر کپی کنید.
۲. روی start-clinic.bat دو بار کلیک کنید (پایتون باید نصب باشد).
۳. مرورگر با آدرس http://localhost:8931/index.html باز می‌شود.
۴. از منوی مرورگر (سه‌نقطه) گزینه زیر را بزنید:
   - کروم: Save and share > Install page as app
     (در نسخه‌های جدید: Install «کلینیک دندانپزشکی»)
   - اج: Apps > Install this site as an app
۵. حالا آیکون «کلینیک دندان» روی دسکتاپ و منوی استارت می‌آید
   و بعد از آن بدون سرور هم از همان آیکون باز می‌شود.

اجرای خودکار با بالا آمدن ویندوز (اختیاری):
۱. کلید Win+R و تایپ shell:startup و Enter
۲. از آیکون نصب‌شده یا فایل start-clinic.bat یک میان‌بر (Shortcut)
   داخل همین پوشه Startup بسازید.

-----------------------------------------------------
۲) اندروید
-----------------------------------------------------
نکته مهم: نصب واقعی (Add to Home screen با حالت اپ) نیاز به
آدرس اینترنتی (https) دارد؛ یعنی پوشه باید روی یک هاست باشد.
ساده‌ترین راه‌های رایگان:
- Netlify Drop: فایل‌ها را در app.netlify.com/drop بکشید و رها کنید
- GitHub Pages / Cloudflare Pages
بعد در گوشی:
۱. آدرس سایت را در Chrome باز کنید.
۲. منوی سه‌نقطه > Add to Home screen (یا Install app).
۳. آیکون «کلینیک دندان» به صفحه اصلی اضافه می‌شود و
   تمام‌صفحه و آفلاین باز می‌شود.

استفاده بدون نصب (سریع):
- فایل index.html را به گوشی منتقل کنید و مستقیم باز کنید؛
  همه امکانات کار می‌کند (فقط آیکون جدا و حالت آفلاین کامل ندارد).

function addAppointment(){
  var iso=apptIso();var sh=SHIFTS.find(x=>x.id===apptShift);
  var doc=document.getElementById("apptDoctor").value;var unit=document.getElementById("apptUnit").value;
  var patient=document.getElementById("apptPatient").value.trim();var time=document.getElementById("apptTime").value;
  var svcName=document.getElementById("apptService").value;
  if(!doc){alert("برای این شیفت پزشکی فعال نیست");return;}
  if(!patient){alert("نام بیمار را وارد کنید");return;}
  if(!time){alert("ساعت پذیرش را وارد کنید");return;}
  var st=timeToMin(time);if(st<sh.start||st>sh.end){alert("ساعت "+sh.label+" فقط "+sh.time+" مجاز است");return;}
  var svc=services.find(s=>s.name===svcName)||{name:svcName,dur:30};
  if(st+svc.dur>sh.end+30){if(!safeConfirm("⚠️ ساعت پایان ("+minToTime(st+svc.dur)+") از ساعت شیفت ("+sh.time+") می‌گذرد. ثبت شود؟"))return;}
  var clash=appointments.find(function(a){if(a.date!==iso||a.shift!==apptShift)return false;if(a.doctor!==doc&&a.unit!==unit)return false;var s2=timeToMin(a.time),e2=s2+(a.dur||30);return Math.max(st,s2)<Math.min(st+svc.dur,e2);});
  if(clash){if(!safeConfirm("⚠️ تداخل: "+clash.doctor+" ("+clash.unit+") از "+clash.time+" تا "+minToTime(timeToMin(clash.time)+(clash.dur||30))+" نوبت «"+clash.patient+"» را دارد. باز هم ثبت شود؟"))return;}
  appointments.push({id:Date.now(),date:iso,shift:apptShift,doctor:doc,unit:unit,patient:patient,time:time,service:svc.name,dur:svc.dur});
  document.getElementById("apptPatient").value="";renderAppointments();autoSave();
}
